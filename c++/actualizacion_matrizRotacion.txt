float dot = v_ini[1]; // porque v_fin = [0,1,0]
float norm_ini = sqrt(v_ini[0]*v_ini[0] + v_ini[1]*v_ini[1] + v_ini[2]*v_ini[2]);

float cos_theta = dot / norm_ini;

// ✅ caso 1: iguales
if (cos_theta > 0.999f) {
    // identidad
    set_identity(rotation_matrix);
    return;
}

// ✅ caso 2: opuestos (Rotación de 180° alrededor de un eje PERPENDICULAR a v_ini, encontrar un vector ortogonal a v_ini.)
if (cos_theta < -0.999f) {
    float axis[3];

    if (fabs(v_ini[0]) < fabs(v_ini[1])) {
        axis[0] = 0;
        axis[1] = -v_ini[2];
        axis[2] = v_ini[1];
    } else {
        axis[0] = -v_ini[2];
        axis[1] = 0;
        axis[2] = v_ini[0];
    }

    float norm = sqrt(axis[0]*axis[0] + axis[1]*axis[1] + axis[2]*axis[2]);
    axis[0] /= norm;
    axis[1] /= norm;
    axis[2] /= norm;

    float q0 = 0;
    float q1 = axis[0];
    float q2 = axis[1];
    float q3 = axis[2];

    quaternion_to_matrix(q0, q1, q2, q3);
    return;
}
